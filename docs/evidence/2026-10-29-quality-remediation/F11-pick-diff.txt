    method any Pick(SkillBook book, PlayerStats st)
        -- 카드 3장: ① 새 스킬(안 배운 메인) ② 쓰던 스킬 랭크 업(칸에 넣은 것 우선) ③ 강화(칸에 넣은 스킬의 강화 우선). 다시 섞으면 같은 종류의 다른 후보
        local news, ranks, nodes = {}, {}, {}
        local slotted = self:Slotted(book)
        for _, id in ipairs(_SkillData:Mains()) do
            if book:CanLearn(id, st.Level, st.SkillPoints) == "" then
                if book:GetLevel(id) == 0 then table.insert(news, id)
                elseif slotted[id] then table.insert(ranks, 1, id)
                else table.insert(ranks, id) end
            end
            for _, nid in ipairs(_SkillData:NodesOf(id)) do
                if book:CanLearn(nid, st.Level, st.SkillPoints) == "" then
                    if slotted[id] then table.insert(nodes, 1, nid) else table.insert(nodes, nid) end
                end
            end
        end
        local function at(list)
            if #list == 0 then return nil end
            return list[(self.Shuffle % #list) + 1]
        end
        local offer = {}
        -- 목록마다 따로 고른다 (예전 { at(news), at(ranks), at(nodes) }는 새 스킬이 없으면 첫 칸이 nil이라 ipairs가 바로 멈춰 카드 0장 — 10-29 통합 메이커)
        for _, list in ipairs({ news, ranks, nodes }) do
            local id = at(list)
            if id ~= nil then table.insert(offer, id) end
        end
        log("[VRF] offer pick new=" .. #news .. " rank=" .. #ranks .. " node=" .. #nodes .. " → " .. table.concat(offer, ","))
        return offer
    end
